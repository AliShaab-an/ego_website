# ==========================================
# PRODUCTION .htaccess FILE
# Upload this to your production server's public folder
# ==========================================

<IfModule mod_rewrite.c>
  RewriteEngine On
  
  # For production (domain root)
  RewriteBase /

  # Allow static assets (CSS, JS, images, fonts) to load directly
  RewriteCond %{REQUEST_URI} \.(js|mjs|css|png|jpg|jpeg|gif|svg|ttf|woff|woff2|ico)$ [NC]
  RewriteRule ^ - [L]

  # Hide .php extension - redirect /page.php to /page
  RewriteCond %{THE_REQUEST} ^GET\ /[^?\s]*\.php [NC]
  RewriteRule ^(.*)\.php$ /$1 [R=301,L,NC]

  # Route /page to /page.php if file exists
  RewriteCond %{REQUEST_FILENAME} !-f
  RewriteCond %{REQUEST_FILENAME} !-d
  RewriteCond %{REQUEST_FILENAME}.php -f
  RewriteRule ^(.*)$ $1.php [L]

  # Admin area routing
  RewriteCond %{REQUEST_URI} ^/admin [NC]
  RewriteCond %{REQUEST_FILENAME} !-f
  RewriteCond %{REQUEST_FILENAME} !-d
  RewriteRule ^ admin/index.php [QSA,L]

  # Frontend routing - everything else goes to index.php
  RewriteCond %{REQUEST_FILENAME} !-f
  RewriteCond %{REQUEST_FILENAME} !-d
  RewriteRule ^ index.php [QSA,L]
</IfModule>

# Security: block backend system folders
<IfModule mod_rewrite.c>
  RewriteRule ^(app|vendor|logs|cache)/ - [R=404,L]
</IfModule>

# Correct MIME types
AddType application/javascript .js
AddType application/javascript .mjs
AddType text/css .css
AddType image/svg+xml .svg

# Browser caching
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresByType image/jpeg "access plus 1 month"
  ExpiresByType image/png  "access plus 1 month"
  ExpiresByType text/css "access plus 1 month"
  ExpiresByType application/javascript "access plus 1 month"
</IfModule>

# GZIP compression
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/plain text/xml text/css application/javascript application/json
</IfModule>
